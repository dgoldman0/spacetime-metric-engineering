% The components of the stress-energy tensor in one observer's frame. The
% figure carries its own reading rule: each row is what is carried (energy,
% or one component of momentum), each column is where it goes (stored in
% place, or flowing along x, y or z), and every cell holds its symbol.
\documentclass[tikz,border=1pt]{standalone}
\input{figpreamble}
\begin{document}
\begin{tikzpicture}[x=1pt, y=1pt,
  cell/.style={rounded corners=2pt, draw=white, line width=1.2pt},
  sym/.style={font=\small, text=black!88},
  idx/.style={font=\small},
  hdr/.style={font=\sffamily\scriptsize, text=black!72, align=center,
    inner sep=0.5pt, anchor=south},
  rowlbl/.style={font=\sffamily\scriptsize, text=black!72, anchor=east,
    inner sep=0.5pt},
  axttl/.style={font=\sffamily\scriptsize\bfseries, text=segray},
  key/.style={rounded corners=1.5pt, minimum width=8pt, minimum height=8pt,
    inner sep=0pt},
  keylbl/.style={font=\sffamily\scriptsize, text=black!85, anchor=west,
    inner sep=0.5pt},
]
\def\cw{34}\def\ch{21}
% Cells, coloured by kind.
\foreach \i in {0,...,3} \foreach \j in {0,...,3} {
  \ifnum\i=0 \ifnum\j=0 \def\c{seorange!45}\else\def\c{seteal!30}\fi
  \else\ifnum\j=0 \def\c{seteal!30}\else\ifnum\i=\j \def\c{seblue!40}%
  \else\def\c{seblue!14}\fi\fi\fi
  \fill[cell, fill=\c] (\j*\cw,-\i*\ch) rectangle ++(\cw,-\ch);
}
% Every component, by name.
\foreach \row [count=\i from 0] in {
    {\rho,S_x,S_y,S_z},
    {S_x,p_x,\sigma_{xy},\sigma_{xz}},
    {S_y,\sigma_{yx},p_y,\sigma_{yz}},
    {S_z,\sigma_{zx},\sigma_{zy},p_z}} {
  \foreach \e [count=\j from 0] in \row {
    \node[sym] at (\j*\cw+0.5*\cw, -\i*\ch-0.5*\ch) {$\e$};
  }
}
% Second index: where it goes.
\foreach \l [count=\j from 0] in {t,x,y,z}
  \node[idx] at (\j*\cw+0.5*\cw, 7) {$\l$};
\node[hdr] at (0.5*\cw, 14) {stored\strut\\here\strut};
\foreach \l [count=\j from 1] in {x,y,z}
  \node[hdr] at (\j*\cw+0.5*\cw, 14) {flowing\strut\\along $\l$\strut};
\node[axttl] at (2*\cw, 42) {second index: where it goes};
% First index: what is carried.
\foreach \l [count=\i from 0] in {t,x,y,z}
  \node[idx] at (-7, -\i*\ch-0.5*\ch) {$\l$};
\foreach \w [count=\i from 0] in {energy,$x$-momentum,$y$-momentum,$z$-momentum}
  \node[rowlbl] at (-15, -\i*\ch-0.5*\ch) {\w};
\node[axttl, rotate=90] at (-66, -2*\ch) {first index: what is carried};
% Key.
\begin{scope}[shift={(-58,-4*\ch-15)}]
  \node[key, fill=seorange!45] at (4,0) {};
  \node[keylbl] at (11,0) {energy density $\rho$};
  \node[key, fill=seteal!30] at (4,-12) {};
  \node[keylbl] at (11,-12) {energy flux $S$, equal to momentum density};
  \node[key, fill=seblue!40] at (140,0) {};
  \node[keylbl] at (147,0) {pressure $p$};
  \node[key, fill=seblue!14] at (140,-12) {};
  \node[keylbl] at (147,-12) {shear stress $\sigma$};
\end{scope}
\end{tikzpicture}
\end{document}
